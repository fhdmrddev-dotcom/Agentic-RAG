**1. Create a `ToolResultBlock` component** inside ToolCallPanel.tsx:

```tsx
function ToolResultBlock({ tc }: { tc: ToolCall }) {
  const [open, setOpen] = useState(false)
  // Parse result defensively
  let parsed: any = null
  try { parsed = tc.result ? JSON.parse(tc.result) : null } catch { /* ignore */ }
  if (!parsed || parsed.error) return parsed?.error ? (
    <div className="mt-1 text-xs text-red-400 italic">{parsed.error}</div>
  ) : null

  const summary = resultSummary(tc.name, parsed)
  return (
    <div className="mt-1.5">
      <button onClick={() => setOpen(v => !v)} className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground/70 transition-colors">
        {open ? <ChevronDown ... /> : <ChevronRight ... />}
        <span>{summary}</span>
      </button>
      {open && <div className="mt-1.5 ml-4">{renderResult(tc.name, parsed)}</div>}
    </div>
  )
}
```

**2. Create `resultSummary(name, parsed)` function** — returns a one-line collapsed summary:
- `ls`: `"{folders.length} folders, {documents.length} documents"` (e.g., "2 folders, 5 documents")
- `tree`: count total nodes recursively, `"{count} items"` (e.g., "12 items")
- `grep`: `"{total} matches"` (e.g., "3 matches")
- `glob`: `"{total} matches"` (e.g., "5 matches")

**3. Create `renderResult(name, parsed)` function** — returns JSX for expanded view:

**ls renderer:**
