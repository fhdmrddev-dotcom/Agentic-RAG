```python
"""Integration test for Phase 058 — cross-tab GET unblocked during SSE streaming.

D-058-09 gate: passes only when the aexec() wrapping is in effect so that
a concurrent GET on Thread B returns within 1.0s while Thread A's SSE stream
is actively in-flight.

Pre-058 behaviour: the sync .execute() inside send_message/event_stream holds
the asyncio event loop; the GET on Thread B queues until the SSE finishes (~2s).
Post-058 behaviour: aexec() dispatches .execute() to a threadpool worker,
freeing the event loop; the GET returns in ~20ms.

Uses httpx.AsyncClient (NOT sync TestClient) because genuine asyncio.create_task
concurrency is required. Sync TestClient runs each request serially.
"""
import asyncio
import time
from unittest.mock import MagicMock, patch
from uuid import uuid4

import httpx
import pytest

# Adjust these imports to match the actual conftest/test_threads export paths
# discovered in Step A above.
from app.main import app


THREAD_A = str(uuid4())
THREAD_B = str(uuid4())
USER_ID = "test-user-058"


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

def _make_result(data):
    """Minimal mock of a supabase APIResponse."""
    m = MagicMock()
    m.data = data
    m.count = len(data) if isinstance(data, list) else None
    return m


def _make_sse_chunk(token: str) -> dict:
    """One delta SSE chunk in the shape create_streaming_chat yields."""
    chunk = MagicMock()
    chunk.choices = [MagicMock()]
    chunk.choices[0].delta = MagicMock()
    chunk.choices[0].delta.content = token
    chunk.choices[0].finish_reason = None
    return chunk


def _make_done_chunk() -> dict:
    """Final SSE chunk with finish_reason='stop'."""
    chunk = MagicMock()
    chunk.choices = [MagicMock()]
    chunk.choices[0].delta = MagicMock()
    chunk.choices[0].delta.content = None
    chunk.choices[0].finish_reason = "stop"
    return chunk


def _slow_chunks():
    """Sync generator that yields 8 tokens with 200ms sleep between each.

    Total stream duration: ~1.8s. Keeps SSE connection open long enough for
    the concurrent GET to race against it.
    """
    for token in ("a", "b", "c", "d", "e", "f", "g", "h"):
        time.sleep(0.2)
        yield _make_sse_chunk(token)
    yield _make_done_chunk()


def _thread_row(thread_id: str = THREAD_A) -> dict:
    return {"id": thread_id, "user_id": USER_ID, "folder_id": None}


def _message_row(role: str = "user", content: str = "hello") -> dict:
    return {
        "id": str(uuid4()),
        "thread_id": THREAD_A,
        "user_id": USER_ID,
        "role": role,
        "content": content,
        "tool_calls": None,
        "created_at": "2026-05-01T00:00:00+00:00",
    }


async def _consume_sse(client: httpx.AsyncClient, thread_id: str) -> None:
    """Open the SSE stream and read until exhausted or cancelled."""
    async with client.stream(
        "POST",
        f"/threads/{thread_id}/messages",
        json={"content": "hello"},
        headers={"Authorization": "Bearer test-token"},
        timeout=30.0,
    ) as r:
        async for _ in r.aiter_lines():
            pass


# ---------------------------------------------------------------------------
# Test
# ---------------------------------------------------------------------------

@pytest.mark.asyncio
async def test_cross_tab_unblocked_during_sse():
    """GET /threads/B/messages returns <1.0s while Thread A's SSE is streaming.

    This is the D-058-09 gate. If it fails, the aexec() wrapping is not in
    effect and the event loop is still blocking on sync .execute() calls.
    """
    # Supabase mock: provide a generous side_effect list covering all DB calls
    # that fire across both threads. Order follows the actual call sequence for
    # Thread A's send_message handler, then Thread B's GET handler. Any extra
    # calls fall through to return_value (empty result).
    mock_execute = MagicMock(
        side_effect=[
            # Thread A — send_message preamble (pre-stream)
            _make_result(_thread_row(THREAD_A)),      # thread ownership SELECT
            _make_result([_message_row()]),            # user message INSERT
            # Thread A — event_stream generator
            _make_result({"folder_id": None}),        # folder scope SELECT (single)
            _make_result([_message_row()]),            # history reconstruction
            _make_result([]),                          # skills catalog
            _make_result([]),                          # user_memory
            # Thread B — GET /threads/B/messages (concurrent, may arrive any time)
            _make_result([]),                          # Thread B messages SELECT
            # Thread A — post-stream
            _make_result([_message_row("assistant", "abcdefgh")]),  # persist assistant
            _make_result([]),                          # updated_at write
        ]
    )
    # Fallback for any additional calls (tool writes, audit, etc.)
    mock_execute.side_effect = list(mock_execute.side_effect) + [_make_result([])] * 20

    mock_supabase = MagicMock()
    mock_supabase.table.return_value.select.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.insert.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.update.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.upsert.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.delete.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.eq.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.or_.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.order.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.limit.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.single.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.in_.return_value = mock_supabase.table.return_value
    mock_supabase.table.return_value.execute = mock_execute
    mock_supabase.rpc.return_value.execute = mock_execute

    mock_user = {"id": USER_ID, "email": "test@example.com"}

    with (
        patch("app.api.threads.create_streaming_chat", return_value=_slow_chunks()),
        patch("app.dependencies.get_supabase", return_value=mock_supabase),
        patch("app.dependencies.get_current_user", return_value=mock_user),
    ):
        async with httpx.AsyncClient(app=app, base_url="http://test") as c:
            # Start the SSE stream on Thread A — keeps the event loop busy
            sse_task = asyncio.create_task(_consume_sse(c, THREAD_A))

            # Give the SSE handler a moment to enter the generator and start
            # blocking on the first slow chunk
            await asyncio.sleep(0.1)

            # Race: GET Thread B's messages while SSE is streaming
            t0 = time.monotonic()
            r = await c.get(
                f"/threads/{THREAD_B}/messages",
                headers={"Authorization": "Bearer test-token"},
            )
            elapsed = time.monotonic() - t0

            # Cancel the SSE task now that we have our measurement
            sse_task.cancel()
            try:
                await sse_task
            except (asyncio.CancelledError, Exception):
                pass

    assert elapsed < 1.0, (
        f"GET /threads/{THREAD_B}/messages took {elapsed:.2f}s while Thread A "
        f"was streaming — aexec() wrapping may not be in effect (pre-058 blocking "
        f"behaviour). Expected < 1.0s."
    )
    assert r.status_code == 200, f"Expected 200, got {r.status_code}: {r.text[:200]}"
```
