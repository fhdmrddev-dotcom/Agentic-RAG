    ```python
    async def _reextract_refill_empty_descriptions(
        *,
        document_id: str,
        user_id: str,
        raw: bytes,
        mime_type: str,
        supabase: Client,
    ) -> DocumentResponse:
        """Phase 072 D-072-04 — refill document_images.description='' rows without
        running the full /reextract delete-cascade.

        Steps:
          1. SELECT empty-description rows older than 5 minutes for this doc.
          2. Re-run the PDF/DOCX image extraction pass on the storage raw bytes
             (we need b64_png to call describe_image; D-04 means b64_png is
             NOT stored in document_images, so this is unavoidable).
          3. Match each empty row's image_index to the freshly-extracted b64.
          4. Call describe_image; if non-empty, UPDATE the row.
          5. Return DocumentResponse with the current doc state.

        Failure modes:
          - No empty rows: return doc state (no-op).
          - Image re-extract fails: log warning, return current state (no UPDATE).
          - describe_image returns '' again: skip UPDATE for that row.
        """
        from app.services.multimodal_service import (  # noqa: PLC0415
            extract_pdf_images,
            extract_docx_images,
            describe_image,
        )
        from app.models.user_settings import load_app_settings  # noqa: PLC0415

        # 1. Fetch empty rows (threadpool-wrapped per D-v2.5-01).
        empties_resp = await run_in_threadpool(
            lambda: supabase.table("document_images")
            .select("id, image_index, page")
            .eq("document_id", document_id)
            .eq("user_id", user_id)
            .eq("description", "")
            .lt("extracted_at", "now() - interval '5 minutes'")
            .execute()
        )
        empty_rows = empties_resp.data or []
        if not empty_rows:
            log.info(
                "retry_empty_descriptions_only: no empty rows for document %s",
                document_id,
            )
            # Return current doc state for response shape.
            doc_resp = await run_in_threadpool(
                lambda: supabase.table("documents")
                .select("*")
                .eq("id", document_id)
                .eq("user_id", user_id)
                .single()
                .execute()
            )
            return DocumentResponse(**doc_resp.data)

        # 2. Re-extract images from storage bytes.
        try:
            if mime_type == "application/pdf":
                fresh_images = extract_pdf_images(raw)
            elif mime_type == "application/vnd.openxmlformats-officedocument.wordprocessingml.document":
                fresh_images = extract_docx_images(raw)
            else:
                fresh_images = []
        except Exception as exc:
            log.warning(
                "retry_empty_descriptions_only: image re-extract failed for %s: %s",
                document_id, exc,
            )
            fresh_images = []

        if not fresh_images:
            log.info(
                "retry_empty_descriptions_only: no images re-extracted for %s; "
                "%d empty rows left untouched",
                document_id, len(empty_rows),
            )
            doc_resp = await run_in_threadpool(
                lambda: supabase.table("documents")
                .select("*")
                .eq("id", document_id)
                .eq("user_id", user_id)
                .single()
                .execute()
            )
            return DocumentResponse(**doc_resp.data)

        fresh_by_index = {img["image_index"]: img for img in fresh_images}
        app_settings = load_app_settings()
        refilled = 0

        # 3 + 4. For each empty row, find matching fresh b64 + describe.
        for row in empty_rows:
            idx = row.get("image_index")
            fresh = fresh_by_index.get(idx)
            if fresh is None:
                continue
            b64 = fresh.get("b64_png")
            if not b64:
                continue
            try:
                desc = describe_image(b64, app_settings)
            except Exception as exc:
                log.debug(
                    "retry_empty_descriptions_only: describe_image failed on row %s: %s",
                    row.get("id"), exc,
                )
                continue
            if not desc:
                continue
            # 5. UPDATE this row (threadpool-wrapped per D-v2.5-01).
            await run_in_threadpool(
                lambda r=row, d=desc: supabase.table("document_images")
                .update({"description": d})
                .eq("id", r["id"])
                .execute()
            )
            refilled += 1

        log.info(
            "retry_empty_descriptions_only: refilled %d of %d empty rows for %s",
            refilled, len(empty_rows), document_id,
        )

        doc_resp = await run_in_threadpool(
            lambda: supabase.table("documents")
            .select("*")
            .eq("id", document_id)
            .eq("user_id", user_id)
            .single()
            .execute()
        )
        return DocumentResponse(**doc_resp.data)
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client→/reextract retry-only | Authenticated request reaches the retry branch; the `retry_empty_descriptions_only` flag is a boolean — no injection surface, but RLS still gates the empty-row SELECT |
| storage→re-extract loop | Storage bytes are re-extracted server-side without the delete cascade; same trust surface as the existing /reextract; no NEW surfaces opened |
| describe_image→vision-LLM | The retry branch invokes the vision LLM with the freshly-extracted b64; same trust surface as the existing extract_and_store_images path |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-072-04-01 | I (Info disclosure) | retry branch SELECT against document_images might leak existence to non-owner | low | mitigate | Owner-only `.eq("user_id", user_id)` predicate preserved on the empty-row SELECT (matches the route-entry owner check at lines 786-799). Even if document_id is correct, mismatched user_id returns empty rows and the helper no-ops. |
| T-072-04-02 | D (Denial of service) | adversarial repeated /reextract?retry_empty_descriptions_only=true calls trigger unbounded vision-LLM calls | medium | mitigate | The 5-minute `extracted_at < now() - interval '5 minutes'` filter prevents thrashing — empty rows just-INSERTed are not retry candidates. Each refilled row removes itself from the retry pool. Multi-user multiplication is bounded by D-072-08's `multimodal_max_vision_calls` cap inherited from extract_and_store_images via describe_image. |
| T-072-04-03 | T (Tampering) | adversarial b64 returned by re-extracted PDF image triggers vision-LLM unsafe-content rejection | low | accept | OpenAI/Anthropic provider handles unsafe content rejection; describe_image catches exceptions and returns ''. The empty row stays empty (next /reextract retry will try again). Operator can manually delete the row via the Documents UI if recurring. |
| T-072-04-04 | E (Elevation of privilege) | retry branch bypasses owner-only delete cascade — could be exploited to refill rows on another user's doc? | low | mitigate | The retry-only branch INHERITS the owner-only SELECT at route entry (documents.py:786). Storage download is owner-gated. Empty-row SELECT is owner-gated. UPDATE is row-id-targeted (matched against the previous SELECT). No cross-user attack surface introduced. |
