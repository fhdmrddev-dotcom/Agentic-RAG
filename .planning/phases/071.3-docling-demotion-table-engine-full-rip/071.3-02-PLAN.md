       ```python
       def gmft_tables(raw: bytes, mime: str) -> list[TableData]:
           """gmft (Microsoft TATR via pypdfium2) table extraction. PDF-only.

           First call lazy-downloads ~120 MB TATR weights to the HF cache —
           expect +15-30s on cold start. Subsequent calls are fast.

           Thread-safety: gmft GitHub issue #55 — moot under single-uvicorn-worker
           (D-v2.5-02) + run_in_threadpool discipline at route level (D-v2.5-01).
           The route handler already wraps extract_composable in run_in_threadpool;
           no second wrap needed inside this adapter.

           Returns list[TableData] with bbox populated (TATR exposes layout coords
           via CroppedTable.bbox in pdfium page coords).

           Plan 071.3 D-071.3-05.
           """
           if mime != PDF_MIME:
               return []
           import os                                                       # noqa: PLC0415
           import tempfile                                                  # noqa: PLC0415
           from gmft.auto import AutoTableDetector, AutoFormatter           # noqa: PLC0415
           from gmft.pdf_bindings.pdfium import PyPDFium2Document           # noqa: PLC0415

           with tempfile.NamedTemporaryFile(suffix=".pdf", delete=False) as tf:
               tf.write(raw)
               tmp_path = tf.name
           try:
               detector = AutoTableDetector()
               formatter = AutoFormatter()
               doc = PyPDFium2Document(tmp_path)
               try:
                   out: list[TableData] = []
                   for page_idx, page in enumerate(doc, start=1):
                       cropped_tables = detector.extract(page)
                       for ti, ct in enumerate(cropped_tables):
                           try:
                               ft = formatter.extract(ct)
                               df = ft.df()
                               headers = [str(c) for c in df.columns.tolist()]
                               rows = [[str(c) for c in row] for row in df.values.tolist()]
                           except Exception as exc:
                               log.warning("gmft_tables: format failed on page %d table %d: %s", page_idx, ti, exc)
                               continue
                           bbox = None
                           if getattr(ct, "bbox", None) is not None:
                               l, t, r, b = ct.bbox
                               bbox = {"l": float(l), "t": float(t), "r": float(r), "b": float(b), "page": page_idx}
                           out.append(TableData(
                               page=page_idx,
                               table_index=ti,
                               headers=headers,
                               rows=rows,
                               bbox=bbox,
                           ))
                   return out
               finally:
                   doc.close()
           finally:
               try:
                   os.unlink(tmp_path)
               except Exception:
                   pass
       ```

       ```python
           Flavor=stream uses PDFMiner whitespace clustering — best for borderless
           academic tables per arXiv 2410.09871 Section 2.3. ~80 MB install.

           Plan 071.3 D-071.3-05.

           if mime != PDF_MIME:

           import os                                                       # noqa: PLC0415
           import tempfile                                                  # noqa: PLC0415

           with tempfile.NamedTemporaryFile(suffix=".pdf", delete=False) as tf:
               tf.write(raw)
               tmp_path = tf.name
           try:
               tables = camelot.read_pdf(tmp_path, flavor="stream", pages="all")
               out: list[TableData] = []
               for ti, t in enumerate(tables):
                   df = t.df
                   headers = [str(c) for c in df.iloc[0].tolist()] if len(df) > 0 else []
                   rows = [[str(c) for c in row] for row in df.iloc[1:].values.tolist()]

                           l, t_, r, b = t._bbox

                           pass
                   out.append(TableData(
                       page=int(t.page) if hasattr(t, "page") else None,
                       table_index=ti,
                       headers=headers,
                       rows=rows,
                       bbox=bbox,
                   ))

           finally:
               try:
                   os.unlink(tmp_path)

                   pass
       ```
