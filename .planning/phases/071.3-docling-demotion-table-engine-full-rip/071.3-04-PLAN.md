```python
def pymupdf_full_images_pdf(raw: bytes) -> list[ImageData]:
    import fitz                                              # noqa: PLC0415
    import base64, io                                        # noqa: PLC0415
    from PIL import Image as PILImage                         # noqa: PLC0415
    doc = fitz.Document(stream=raw, filetype="pdf")
    try:
        images: list[ImageData] = []
        for page_num, page in enumerate(doc, start=1):
            for img_idx, img_info in enumerate(page.get_images(full=True)):
                xref = img_info[0]
                pix = fitz.Pixmap(doc, xref)
                if pix.alpha:
                    pix = fitz.Pixmap(fitz.csRGB, pix)
                buf = io.BytesIO()
                pix.pil_save(buf, format="PNG")
                images.append(ImageData(
                    page=page_num,
                    image_index=img_idx,
                    data=base64.b64encode(buf.getvalue()).decode("ascii"),
                    mime_type="image/png",
                    bbox=None,  # page.get_images doesn't expose bbox directly
                ))
                pix = None
        return images
    finally:
        doc.close()
```
