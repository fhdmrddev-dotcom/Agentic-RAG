```python
# Phase 071.1 Layer 2 (D-071.1-02) — wall-clock fail-safe for Docling stalls
# that wedge in native code (TableFormer/layout). Layer 1 in docling.py:50
# checks at batch boundaries only — insufficient for native wedges.
docling_timeout_s = float(os.getenv("EXTRACTOR_DOCLING_TIMEOUT_S", "120"))
wall_clock_s = docling_timeout_s + 10  # +10s buffer over Docling's own check

from app.services.extraction_service import get_extractor, ExtractionError  # noqa: PLC0415
mime_type = target["mime_type"]
extract_start = time.perf_counter()
extracted_doc: ExtractedDocument | None = None
engine_used: str = body.engine
docling_error: str | None = None
docling_duration_ms: int = 0

try:
    extractor = get_extractor(mime_type, engine_override=body.engine)
    if extractor is None:
        raise HTTPException(status_code=422, detail=f"No extractor for {mime_type!r}")

    try:
        extracted_doc = await asyncio.wait_for(
            run_in_threadpool(extractor.extract, raw, mime_type),
            timeout=wall_clock_s,
        )
        text = extracted_doc.text

    except asyncio.TimeoutError:
        # D-071.1-04 — fallback to PyMuPDF on Docling timeout ONLY.
        if body.engine != "docling":
            # Non-Docling engine timed out — fail loud, don't auto-fallback.
            raise HTTPException(status_code=422,
                detail=f"engine={body.engine} timed out after {wall_clock_s}s")

        docling_duration_ms = int((time.perf_counter() - extract_start) * 1000)
        docling_error = f"docling timeout after {wall_clock_s}s (Layer 2 wall-clock)"

        # First telemetry row (Docling failed): inline write via _write_extraction_run_row helper.
        await run_in_threadpool(
            _write_extraction_run_row,
            supabase, document_id, current_user["id"],
            "docling", docling_duration_ms, 0, 0, docling_error,
        )

        # Attempt PyMuPDF — PYMUPDF_TIMEOUT_S enforced inside subprocess fence.
        pymupdf_extractor = get_extractor(mime_type, engine_override="pymupdf")
        if pymupdf_extractor is None:
            raise HTTPException(status_code=422,
                detail=f"Docling timed out and PyMuPDF unavailable for {mime_type!r}")

        engine_used = "pymupdf-fallback"
        try:
            extracted_doc = await run_in_threadpool(
                pymupdf_extractor.extract, raw, mime_type,
            )
            text = extracted_doc.text
        except Exception as fallback_exc:
            # Both engines failed — 422 + second telemetry row.
            await run_in_threadpool(
                _write_extraction_run_row,
                supabase, document_id, current_user["id"],
                "pymupdf-fallback", 0, 0, 0, str(fallback_exc)[:500],
            )
            raise HTTPException(status_code=422,
                detail=f"Docling timed out; PyMuPDF fallback also failed: {fallback_exc}")

except HTTPException:
    raise
except Exception as e:
    raise HTTPException(status_code=422,
        detail=f"Could not extract text via engine={body.engine}: {e}")

extract_duration_ms = int((time.perf_counter() - extract_start) * 1000)
```

```python
def _write_extraction_run_row(
    supabase: Client,
    document_id: str,
    user_id: str,
    engine: str,
    duration_ms: int,
    table_count: int,
    image_count: int,
    error: str | None,
) -> None:
    """Inline writer for pdf_extraction_runs from /reextract fallback path (Phase 071.1 D-071.1-04).

    The happy-path writer lives in ingest_document at line ~893; this helper is used ONLY
    by the /reextract route's exception handler when fallback fires.

    Telemetry-write failures are swallowed (T-071-02-07 pattern) — a telemetry hiccup must
    NOT block the user's extraction outcome.
    """
    try:
        supabase.table("pdf_extraction_runs").insert({
            "document_id": document_id,
            "user_id": user_id,
            "engine": engine,
            "started_at": datetime.now(timezone.utc).isoformat(),
            "duration_ms": duration_ms,
            "table_count": table_count,
            "image_count": image_count,
            "error": error[:1000] if error else None,
        }).execute()
    except Exception as exc:
        import logging  # noqa: PLC0415
        logging.getLogger(__name__).warning(
            "pdf_extraction_runs INSERT failed (engine=%s, doc=%s, from /reextract fallback): %s",
            engine, document_id, exc,
        )
```

```bash
# EXTRACTOR_DOCLING_TIMEOUT_S: Docling per-document timeout in seconds (D-071.1-02).
# Drives both (a) PdfPipelineOptions.document_timeout (checked at batch boundaries inside
# the pipeline) AND (b) the wall-clock asyncio.wait_for fail-safe at the /reextract route
# (which adds a +10s buffer over this value). The wall-clock layer catches Docling stalls
# that wedge in native code (e.g., TableFormer/layout-prediction) before Docling itself
# can check its own timeout. Trade-off (D-v2.5-02 single-worker): the wall-clock layer
# cannot kill the underlying thread — one stuck thread per stuck call until uvicorn restart.

# EXTRACTOR_DOCLING_DISABLE_TABLE_STRUCTURE: bool — when truthy, sets PdfPipelineOptions.
# do_table_structure=False. Lets operators fall back to "pdfium-only" mode on PDFs that
# wedge in TableFormer. Trade-off: tables become text-only chunks instead of structured
# TableData rows. Default false (full TableFormer enabled). Truthy values: 1, true, yes.
EXTRACTOR_DOCLING_DISABLE_TABLE_STRUCTURE=false

# EXTRACTOR_DOCLING_IMAGES_SCALE: float — PdfPipelineOptions.images_scale. Lower values
# (e.g., 1.0 or 0.5) reduce memory pressure on big PDFs by producing smaller rendered
# page images for figure extraction. Default 2.0 (matches today's hardcoded value).
EXTRACTOR_DOCLING_IMAGES_SCALE=2.0
```

## Shared Patterns

### Pattern S1: `run_in_threadpool` wrapping of sync supabase-py builder chains
**Source:** Phase 061+ work + the existing 9116c2b inline fix at `documents.py:609`
**Apply to:** every `.execute()`-terminated builder chain inside the `/reextract` async handler (6 call sites for D-071.1-01)
**Wrapper form:** `await run_in_threadpool(lambda: <builder-chain>.execute())` — RESEARCH.md §7 Option A
**Exception:** storage download is a bound method — pass directly as `await run_in_threadpool(supabase.storage.from_("documents").download, target["file_path"])` (no lambda needed)
**Why lambda not bare `.execute`:** RESEARCH.md Pitfall 1 — the builder chain itself (`.table().select().eq()...`) is pure-Python and fast, so it works either way, but the lambda form makes the wrapping intent visible at the call site.

### Pattern S2: Module-level singleton + env-var reads under lock
**Source:** `extraction_service.py:215-228` (`_read_primary()`) + `docling.py:31-64` (`_get_converter()`)
**Apply to:** the three new Docling env vars in `_get_converter()`
**Convention:** reads happen inside `_CONVERTER_LOCK` at instantiation time, frozen for process lifetime. No separate cache variable needed (singleton itself is the cache). Match the *try/except ValueError + log.warning + default-fallback* shape for numeric parses (RESEARCH.md Pitfall 4).

### Pattern S3: Telemetry-write-swallow (T-071-02-07)
**Source:** `documents.py:888-907` + `documents.py:921-936` (both inside `ingest_document`)
**Apply to:** the new `_write_extraction_run_row` helper for the fallback path
**Convention:** Telemetry insert wrapped in `try/except Exception: log.warning(...)` — a Supabase write hiccup must NEVER block the user's extraction outcome. The catch is broad (`Exception`) because telemetry is best-effort observability, not a contract.

### Pattern S4: Owner-only RLS at the controller boundary
**Source:** `documents.py:554-567` (the existing owner-SELECT block in `/reextract`)
**Apply to:** **PRESERVE VERBATIM** — 071.1 does NOT touch the auth surface. T-071-04-01 mitigation (404 not 403) stays intact through the threadpool sweep; only the `.execute()` call gets `run_in_threadpool`-wrapped.

### Pattern S5: Pydantic `Literal` validation for body schema
**Source:** `documents.py:24-32` (`ReextractRequest.engine: Literal["docling", "pymupdf", "legacy"]`)
**Apply to:** **PRESERVE VERBATIM** — T-071-04-02 mitigation. 071.1 does NOT add a new request field or expand the engine literal; `engine_used` is computed server-side at the route, not received from the client.

### Pattern S6: Lazy heavy imports
**Source:** `docling.py:43-45` (lazy `from docling.document_converter import ...` inside `_get_converter()`) + the existing route-local lazy imports at `documents.py:601-602`
**Apply to:** the new optional `ConversionStatus` import inside `DoclingExtractor.extract()` (if Plan 01 includes the PARTIAL_SUCCESS detection enhancement)

### Pattern S7: Test sequencing via `mock_builder.execute.side_effect` queue
**Source:** `test_documents.py:426-432` (existing happy-path sequence)
**Apply to:** all new `TestReextractDocument` tests. Each test must supply a side_effect list matching the EXACT order of `.execute()` calls the route makes — including the new telemetry INSERT calls from `_write_extraction_run_row`. The sequence for the fallback-fired case is: owner SELECT → delete chunks → delete tables → delete images → UPDATE documents → telemetry row 1 (docling failed) → (PyMuPDF call — no DB write here, the second telemetry row is written by `ingest_document` post-response if it succeeds, or by the inline writer if it fails).
