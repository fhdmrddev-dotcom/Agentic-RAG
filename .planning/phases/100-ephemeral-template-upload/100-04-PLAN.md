2. Add a module-level `validate_ooxml(filename, raw) -> str` helper (RESEARCH Pattern 2 verbatim):
```python
_ALLOWED_EXT = {".docx", ".pptx", ".xlsx"}
_OOXML_MARKER = {".docx": "word/", ".pptx": "ppt/", ".xlsx": "xl/"}

def validate_ooxml(filename: str, raw: bytes) -> str:
    ext = "." + filename.rsplit(".", 1)[-1].lower() if "." in filename else ""
    if ext not in _ALLOWED_EXT:
        raise HTTPException(422, f"Unsupported type {ext}. Allowed: .docx, .pptx, .xlsx")
    bio = io.BytesIO(raw)
    if not zipfile.is_zipfile(bio):
        raise HTTPException(422, "File is not a valid Office document (not a ZIP/OOXML container)")
    with zipfile.ZipFile(bio) as zf:
        names = zf.namelist()
        if "[Content_Types].xml" not in names:
            raise HTTPException(422, "File is not a valid OOXML document")
        if not any(n.startswith(_OOXML_MARKER[ext]) for n in names):
            raise HTTPException(422, f"File contents do not match a {ext} document")
    return ext
```

3. Add the POST route (lands HERE in workspace.py — NOT threads.py, per G-5):
```python
@router.post("/files")
async def upload_workspace_template(
    thread_id: str,
    file: UploadFile = File(...),
    current_user: dict = Depends(get_current_user),
    supabase: Client = Depends(get_supabase),
):
    await _verify_thread_ownership(thread_id, current_user, supabase)  # 404 on non-owner (SC#1 RLS half)
    raw = await file.read()
    if len(raw) == 0:
        raise HTTPException(422, "File is empty")
    if len(raw) > 10 * 1024 * 1024:
        raise HTTPException(422, "File too large. Maximum size is 10 MB.")
    ext = validate_ooxml(file.filename or "", raw)            # D-12 magic-byte gate
    ttl_hours = (await load_app_settings_async()).template_ttl_hours   # D-05
    expires_at = datetime.now(timezone.utc) + timedelta(hours=ttl_hours)
    safe_name = (file.filename or f"template{ext}").replace("/", "_").replace("\\", "_")
    path = f"/{uuid4().hex[:8]}-{safe_name}"
    try:
        result = await ws_write_file(
            await get_pg_pool(), supabase,
            thread_id=UUID(thread_id), user_id=UUID(current_user["id"]),
            path=path, content=raw,
            kind="template_input", expires_at=expires_at,    # D-12 / D-05
        )
    except FileTooLargeError as e:
        raise HTTPException(422, str(e))
    except WorkspaceError as e:
        raise HTTPException(422, str(e))
    result["kind"] = "template_input"
    result["expires_at"] = expires_at.isoformat()
    return result
```
The uuid path prefix prevents duplicate-filename collisions on the `(thread_id, path)` unique key (Discretion: new file, same expiry rules). NO SSE emit (RESEARCH Open-Q3 — the handler has no run_id/ctx.redis; the frontend reconciles by upserting the returned row in Plan 06). `validate_path` inside `write_file` still applies to the generated path.
