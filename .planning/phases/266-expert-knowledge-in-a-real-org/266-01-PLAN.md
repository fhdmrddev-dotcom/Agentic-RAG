  async def get_expert_install(pool, *, org_id: UUID, bundle_id: UUID) -> dict | None
      SELECT i.*, (f.id IS NOT NULL) AS folder_exists FROM public.expert_installs i
      LEFT JOIN public.folders f ON f.id = i.folder_id AND f.org_id = i.org_id
      WHERE i.org_id = $1 AND i.expert_bundle_id = $2
  async def list_expert_installs_for_org(pool, *, org_id: UUID) -> list[dict]
      same columns + b.name AS expert_name, b.slug AS expert_slug (JOIN public.expert_bundles b ON b.id = i.expert_bundle_id), WHERE i.org_id = $1
  async def claim_expert_install(pool, *, org_id: UUID, bundle_id: UUID, installed_by: UUID, corpus_version: str) -> dict | None
      INSERT ... VALUES ($1,$2,$3,$4,'installing') ON CONFLICT (org_id, expert_bundle_id) DO UPDATE
      SET status='installing', corpus_version=EXCLUDED.corpus_version, error=NULL, updated_at=now()
      WHERE public.expert_installs.status <> 'installing' OR public.expert_installs.updated_at < now() - interval '10 minutes'
      RETURNING *      — None returned = claim lost (NOT an error). installed_by is NOT overwritten on conflict.
  async def set_expert_install_folder(pool, *, org_id: UUID, bundle_id: UUID, folder_id: UUID, installed_by: UUID) -> dict | None
      UPDATE ... SET folder_id=$3, installed_by=$4, updated_at=now() WHERE org_id=$1 AND expert_bundle_id=$2 RETURNING *
  async def set_expert_install_status(pool, *, org_id: UUID, bundle_id: UUID, status: str, error: str | None) -> dict | None
      UPDATE ... SET status=$3, error=$4, updated_at=now() WHERE org_id=$1 AND expert_bundle_id=$2 RETURNING *
  async def list_install_corpus_documents(pool, *, org_id: UUID, folder_ids: list[UUID], filenames: list[str]) -> list[dict]
      SELECT id, folder_id, filename, status, chunk_count, error_message, user_id, file_path, mime_type
      FROM public.documents WHERE org_id = $1 AND folder_id = ANY($2::uuid[]) AND is_latest = true AND filename = ANY($3::text[])
