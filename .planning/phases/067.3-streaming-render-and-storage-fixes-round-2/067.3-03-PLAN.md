<objective>
Fix R-1: cross-thread switch mid-stream loses the streamed-into thread's render. Phase 067.2 Plan 01's `useLayoutEffect` synchronizes `activeThreadIdRef` on mount/`thread.id` change, but with TWO concurrent runs (Thread A streaming + Thread B starts), `streamingThreadIdRef` is locked to whichever run is most recent — when the user returns to A while B is still streaming, A's deltas are still arriving but `guardedSetMessages` no-ops them because `streamingThreadIdRef === B.id` while `activeThreadIdRef === A.id`. Per D-067.3-R1-01 (locked in /gsd-discuss-phase 067.3), fix with the **structural per-thread store** that mirrors ChatGPT/Claude.ai client-side architecture: `messagesByThread: Map<string, Message[]>`. Deltas write to their thread's bucket regardless of viewing thread; the visible `messages` array is derived from the bucket of the currently-viewed thread.

Purpose: Bring client-side state shape into alignment with production-grade chat UIs and unblock v2.6+ multi-tenant + org-shared threads (per `project_target_scale.md` + `project_org_level_deferred.md`). The tactical buffer alternative would force a re-do under multi-tenant; the structural fix composes naturally.

    // setMessages calls. ThreadId resolves at the CALL SITE — sendMessage and
    // reconcile have `threadId` in scope; makeStreamCallbacks accepts threadId
    // as a constructor arg.
    const setMessagesForThread = useCallback(
      (threadId: string, updater: Message[] | ((prev: Message[]) => Message[])) => {
        setMessagesByThread((store) => {
          const next = new Map(store)
          const prev = next.get(threadId) ?? []
          const updated = typeof updater === "function"
            ? (updater as (p: Message[]) => Message[])(prev)
            : updater
          next.set(threadId, updated)
          return evictIfOverCapacity(next, threadId)
        })
      },
      [],
    )

    // D-067.3-R1-03: pin streaming + active + just-written buckets; evict LRU
    // among the rest when over CAP. If all over-cap entries are protected,
    // accept temporary overage; will re-check on next write.
    const evictIfOverCapacity = (
      store: Map<string, Message[]>,
      justWrittenThreadId: string,
    ): Map<string, Message[]> => {
      const CAP = 5
      if (store.size <= CAP) return store
      const protectedIds = new Set<string>()
      protectedIds.add(justWrittenThreadId)
      if (streamingThreadIdRef.current) protectedIds.add(streamingThreadIdRef.current)
      if (activeThreadIdRef.current) protectedIds.add(activeThreadIdRef.current)
      let lruId: string | null = null
      let lruTs = Number.POSITIVE_INFINITY
      for (const id of store.keys()) {
        if (protectedIds.has(id)) continue
        const ts = lastViewedAtRef.current.get(id) ?? 0
        if (ts < lruTs) { lruTs = ts; lruId = id }
      }
      if (lruId) {
        const next = new Map(store)
        next.delete(lruId)
        lastViewedAtRef.current.delete(lruId)
        return next
      }
      return store
    }
