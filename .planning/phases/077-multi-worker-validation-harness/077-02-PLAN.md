```python
import asyncio
import json
import os
import socket
import subprocess
import sys
import time
from pathlib import Path
from uuid import uuid4

import asyncpg
import httpx
import pytest
import redis.asyncio as aioredis

_POSTGRES_TEST_DSN = os.environ.get(
    "POSTGRES_DSN",
    "postgresql://postgres:postgres@127.0.0.1:54322/postgres",
)
_REDIS_TEST_URL = os.environ.get("REDIS_URL", "redis://127.0.0.1:6379/0")

def _check_infra_available() -> bool:
    """Check that Redis + Postgres + Docker are reachable."""
    import asyncio as _a
    async def _probe():
        # Redis
        try:
            r = aioredis.from_url(_REDIS_TEST_URL)
            await asyncio.wait_for(r.ping(), timeout=2.0)
            await r.aclose()
        except Exception:
            return False
        # Postgres
        try:
            conn = await asyncio.wait_for(asyncpg.connect(_POSTGRES_TEST_DSN), timeout=2.0)
            await conn.close()
        except Exception:
            return False
        return True
    try:
        loop = _a.new_event_loop()
        try:
            return loop.run_until_complete(_probe())
        finally:
            loop.close()
    except Exception:
        return False

INFRA_AVAILABLE = _check_infra_available()
pytestmark = pytest.mark.skipif(
    not INFRA_AVAILABLE,
    reason="Redis + Postgres infrastructure not reachable; skipping multi-worker tests",
)
```

**Test function `test_50_run_load`:**
Fire 50 concurrent POST requests to `/threads/{thread_id}/messages` using `asyncio.gather`. Each request sends `{"content": f"test message {i}"}` with a 60s timeout. Assert:
- All 50 responses have status 200 (SSE stream started)
- Consume each SSE response until `stream_end` or connection close
- After all 50 complete, query Redis: `await redis.zcard("runs:active")` == 0 (all cleaned up)
- Query Postgres: `SELECT count(*) FROM runs WHERE thread_id = $1 AND status = 'completed'` == 50

**Test function `test_concur01_multi_worker`:**
Adapted from test_058's pattern but against the live subprocess:
- Start an SSE stream on Thread A (POST a message, consume the response stream)
- While streaming, fire a GET `/threads/{thread_id}/messages` on the same thread
- Assert the GET returns in <2.0s (relaxed from 1.0s due to subprocess overhead)
- Cancel the SSE task

**Test function `test_singleton_no_crosstalk`:**
After the 50-run load completes:
- Assert `redis.zcard("runs:active")` == 0 (no leaked entries from either worker)
- Assert `redis.zcard(f"runs_by_thread:{thread_id}")` == 0 (all per-thread entries cleaned)
- Verify no duplicate run_ids exist in any Redis sorted set
