   ```python
   async def resolve_project_subtree(project_folder_id, *, supabase, user_id) -> list[str] | None:
       if project_folder_id is None:
           return None                                    # unbound workflow → whole-KB (unchanged behavior)
       root = str(project_folder_id)
       folders = await fetch_visible_folders(supabase, user_id)   # owner-scoped fetch
       def _walk(rid: str) -> list[str]:
           out = [rid]
           for f in folders:
               if f["parent_id"] == rid:
                   out.extend(_walk(f["id"]))
           return out
       return _walk(root)                                 # list[str] — Pitfall 1: NEVER a set (supabase-py json.dumps raises on a set)
   ```
   Return a `list[str]` (the RPC param channel is JSON-serialized; a real `set` would raise). `None` in → `None` out so an unbound workflow keeps whole-KB behavior.
