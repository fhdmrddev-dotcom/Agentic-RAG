<details><summary>classifier source (re-runnable; reads stdin, prints total/comment/code/blank)</summary>

```js
function classify(source) {
  // Drop a single trailing newline so the count equals `wc -l` on a newline-terminated file.
  const normalized = source.replace(/\r\n/g, "\n").replace(/\n$/, "");
  const lines = normalized === "" ? [] : normalized.split("\n");
  let blank = 0, comment = 0, code = 0, inBlock = false;
  for (const line of lines) {
    const t = line.trim();
    if (inBlock) {
      const closeIdx = line.indexOf("*/");
      if (closeIdx === -1) { comment++; continue; }
      inBlock = false;
      const after = line.slice(closeIdx + 2).replace(/^\}/, "").trim();
      if (after === "") comment++; else code++;
      continue;
    }
    if (t === "") { blank++; continue; }
    if (t.startsWith("//")) { comment++; continue; }
    if (t.startsWith("/*") || t.startsWith("{/*")) {
      const openIdx = line.indexOf("/*");
      const closeIdx = line.indexOf("*/", openIdx + 2);
      if (closeIdx === -1) { inBlock = true; comment++; continue; }
      const after = line.slice(closeIdx + 2).replace(/^\}/, "").trim();
      if (after === "") comment++; else code++;
      continue;
    }
    code++;
  }
  return { total: lines.length, comment, code, blank };
}
```
