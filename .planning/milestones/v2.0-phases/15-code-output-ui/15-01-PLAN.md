// onCodeStdout
(content: string) => {
  setMessages((prev) =>
    prev.map((m) => {
      if (m.id !== assistantId) return m
      const updated = (m.tool_calls ?? []).map((tc) =>
        tc.name === "execute_code" && tc.status === "running"
          ? { ...tc, outputLines: [...(tc.outputLines ?? []), { kind: "stdout" as const, content }] }
          : tc
      )
      return { ...m, tool_calls: updated }
    })
  )
},
