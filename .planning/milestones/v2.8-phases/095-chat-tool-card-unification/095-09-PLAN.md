def _select_hero_filenames(files: list[dict], user_message: str | None) -> set[str]:
    if not files:
        return set()
    declared = {f["filename"] for f in files if (f.get("is_hero") or f.get("hero")) and f.get("filename")}
    if declared:
        return declared                                   # may be multi — collapse to one (see action)
    msg = (user_message or "").lower()
    requested = {ext for ext in _HERO_REQUESTABLE_EXTS if ext in msg}
    if requested:
        matched = {                                       # <-- MULTI-ELEMENT SET — the leak
            f["filename"] for f in files
            if "." in f["filename"] and f["filename"].rsplit(".", 1)[-1].lower() in requested
        }
        if matched:
            return matched                                # <-- return ALL matches — BUG
    hero = max(files, key=lambda f: (int(f.get("size") or 0), int(f.get("iteration") or 0)))
    return {hero["filename"]}                              # fallback already single
