```python
async def _stream_one_iteration(
    *,
    messages: list[dict],
    tools: list[dict],
    model: str,
    user_settings: Any,
    provider: str,
    structured_injected: list[bool],   # single-element inject-once flag (Pitfall 2)
) -> tuple[str, list[dict]]:
    from app.services.provider_gateway import GatewayRequest, open_stream
    from app.services.openai_service import CallingMode

    # Route key: anthropic/google use active_provider; the openai-compat else uses
    # the REGISTRY-derived provider for this model (agent_loop.py:1616-1617) so the
    # adapter's <think>/usage/boundary logic keys correctly. Sub-agents inherit the
    # parent provider, so `provider` IS the active provider here.
    _adapter_provider = provider
    if provider not in ("anthropic", "google"):
        from app.services.openai_service import get_model_capability_async  # or sync equiv
        _cap = await get_model_capability_async(model) or {}
        _adapter_provider = (_cap.get("provider") or provider or "unknown").lower()

    _gw_request = GatewayRequest(
        messages=messages,
        model=model,
        active_provider_name=_adapter_provider,
        tools=tools,
        user_settings=user_settings,
        tool_choice="auto",
    )
    stream, calling_mode = await open_stream(provider, _gw_request)

    # D-03 half b — STRUCTURED inject ONCE (Pitfall 2), before the drain, into the
    # system message. The gateway's openai_compat adapter SKIPS tool emission on
    # STRUCTURED and does NOT inject (openai_compat.py:306) — the consumer must.
    if calling_mode == CallingMode.STRUCTURED and not structured_injected[0]:
        from app.services.agent_loop import TOOL_USAGE_INSTRUCTIONS, _format_tool_list
        from app.services.openai_service import get_tools as _get_tools
        _tl = _format_tool_list(tools if tools is not None else _get_tools(user_settings))
        for _i, _m in enumerate(messages):
            if _m.get("role") == "system":
                messages[_i] = {"role": "system",
                                "content": _m["content"] + TOOL_USAGE_INSTRUCTIONS.format(tool_list=_tl)}
                structured_injected[0] = True
                break

    _build_from_progress = provider not in ("anthropic", "google")  # openai-compat builds from progress

    def _drain() -> tuple[str, list[dict]]:
        # 🔴 IN-05: SYNC generator — `for event in stream:`, NEVER `async for`.
        content_parts: list[str] = []
        buffer: dict[int, dict] = {}
        try:
            for event in stream:
                et = event.get("type")
                if et == "delta":
                    content_parts.append(event.get("content", ""))
                elif et == "reasoning_delta":
                    pass  # sub-agent ignores reasoning text (not round-tripped here)
                elif et == "tool_preparing":
                    if _build_from_progress:
                        idx = event.get("index", len(buffer))
                        buffer.setdefault(idx, {"id": event.get("id", "") or "", "name": event.get("name", "") or "", "arguments": ""})
                elif et == "tool_args_progress":
                    if _build_from_progress:
                        tidx = event["tool_index"]
                        b = buffer.setdefault(tidx, {"id": "", "name": "", "arguments": ""})
                        if event.get("name"):
                            b["name"] = event["name"]
                        b["arguments"] = event.get("code_so_far", "")
                elif et == "tool_start":
                    buffer[len(buffer)] = {"id": event["id"], "name": event["name"], "arguments": json.dumps(event.get("args", {}))}
                # usage/finish ignored (no token-SUM needed for the sub-agent return)
        finally:
            close = getattr(stream, "close", None)
            if close is not None:
                try:
                    close()
                except Exception:  # noqa: BLE001
                    logger.debug("stream close failed; ignoring", exc_info=True)
        return "".join(content_parts), [buffer[i] for i in sorted(buffer)]

    content, tool_calls = await run_in_threadpool(_drain)

    # D-03 half b — STRUCTURED post-parse AFTER the drain (agent_loop.py:1675-1696).
    if calling_mode == CallingMode.STRUCTURED:
        from app.services.tool_parser import parse_structured_tool_calls
        structured = parse_structured_tool_calls(content)
        if structured:
            tool_calls = [{"id": c.id, "name": c.function.name, "arguments": c.function.arguments} for c in structured]
            content = ""

    return content, tool_calls
```
