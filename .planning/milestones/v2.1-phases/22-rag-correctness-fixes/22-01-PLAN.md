```python
"""Tests for RAG correctness fixes: read_document cap, metadata case normalization."""
import pytest


# ── metadata_filter lowercasing ────────────────────────────────────────────────

def _normalize_filter(metadata_filter: dict | None) -> dict | None:
    """Replicate the normalization logic from retrieval_service.search_documents."""
    if metadata_filter:
        return {k: v.lower() if isinstance(v, str) else v for k, v in metadata_filter.items()}
    return metadata_filter


def test_filter_value_lowercased():
    result = _normalize_filter({"document_type": "PDF"})
    assert result == {"document_type": "pdf"}


def test_filter_language_lowercased():
    result = _normalize_filter({"language": "English"})
    assert result == {"language": "english"}


def test_filter_mixed_case():
    result = _normalize_filter({"document_type": "Report", "language": "French"})
    assert result == {"document_type": "report", "language": "french"}


def test_filter_none_passthrough():
    assert _normalize_filter(None) is None


def test_filter_non_string_value_unchanged():
    result = _normalize_filter({"count": 5})
    assert result == {"count": 5}


# ── metadata ingest lowercasing ────────────────────────────────────────────────

def test_document_type_lowercased_at_ingest():
    """Ingest normalization: document_type is lowercased."""
    metadata_dict = {"document_type": "PDF", "title": "Test"}
    if "document_type" in metadata_dict and metadata_dict["document_type"]:
        metadata_dict["document_type"] = metadata_dict["document_type"].lower()
    assert metadata_dict["document_type"] == "pdf"
    assert metadata_dict["title"] == "Test"  # other fields unchanged


def test_language_lowercased_at_ingest():
    """Ingest normalization: language is lowercased."""
    metadata_dict = {"language": "English"}
    if "language" in metadata_dict and metadata_dict["language"]:
        metadata_dict["language"] = metadata_dict["language"].lower()
    assert metadata_dict["language"] == "english"


def test_missing_document_type_no_error():
    """No KeyError when document_type not in metadata_dict."""
    metadata_dict = {"title": "Test"}
    if "document_type" in metadata_dict and metadata_dict["document_type"]:
        metadata_dict["document_type"] = metadata_dict["document_type"].lower()
    assert "document_type" not in metadata_dict


# ── read_document cap ──────────────────────────────────────────────────────────

def test_read_document_cap_logic():
    """read_document ctx_limit uses _CTX_LIMIT_DEFAULT, not len(content)."""
    _CTX_LIMIT_DEFAULT = 3000
    _CTX_LIMIT_SUBAGENT = 10000

    tool_name = "read_document"
    ctx_content = "x" * 5000  # 5000 char document

    # New logic — read_document uses DEFAULT cap
    ctx_limit = (
        _CTX_LIMIT_SUBAGENT if tool_name == "analyze_document"
        else _CTX_LIMIT_DEFAULT
    )
    assert ctx_limit == 3000
    assert len(ctx_content) > ctx_limit  # would have been uncapped before


def test_analyze_document_uses_subagent_cap():
    """analyze_document uses larger _CTX_LIMIT_SUBAGENT cap."""
    _CTX_LIMIT_DEFAULT = 3000
    _CTX_LIMIT_SUBAGENT = 10000
    tool_name = "analyze_document"
    ctx_limit = _CTX_LIMIT_SUBAGENT if tool_name == "analyze_document" else _CTX_LIMIT_DEFAULT
    assert ctx_limit == 10000
```
