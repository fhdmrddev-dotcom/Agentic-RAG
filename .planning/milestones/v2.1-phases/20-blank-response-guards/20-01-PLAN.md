def test_skill_row_none_returns_error_json():

    if not skill_row:

# ── Guard logic unit tests ─────────────────────────────────────────────────────

def test_fallback_triggers_when_full_content_empty():
    """Guard: empty full_content after loop should trigger fallback."""
    full_content = ""
    fallback_triggered = False
    if not full_content:
        fallback_triggered = True
        full_content = "fallback message"
    assert fallback_triggered
    assert full_content == "fallback message"


def test_fallback_does_not_trigger_when_content_exists():
    """Guard: non-empty full_content does not trigger fallback."""
    full_content = "Some real answer."
    fallback_triggered = False
    if not full_content:
        fallback_triggered = True
    assert not fallback_triggered


def test_length_with_tool_calls_triggers_tool_guard():
    """When finish_reason=length and tool_calls_buffer non-empty, tool guard fires."""
    finish_reason = "length"
    tool_calls_buffer = {0: {"id": "tc1", "name": "search_documents", "arguments": '{"q'}}
    guard_fired = False
    if finish_reason == "length" and tool_calls_buffer:
        guard_fired = True
    assert guard_fired

    """When finish_reason=length and tool_calls_buffer empty, text truncation fires."""
    finish_reason = "length"

    text_guard_fired = False
    tool_guard_fired = False
    if finish_reason == "length" and tool_calls_buffer:
        tool_guard_fired = True
    elif finish_reason == "length":
        text_guard_fired = True
    assert text_guard_fired
    assert not tool_guard_fired
