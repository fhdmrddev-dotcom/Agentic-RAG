       ```python
       async def _drain_stream_with_close_on_cancel(stream, timeout_seconds, on_chunk_async):
           """Phase 067.1 Plan 01 Track A: own the stream iteration so the implicit
           for-loop cleanup (which sends GeneratorExit into _TracedStream.__iter__
           and is recorded by langsmith's BaseException catch as the trace error)
           never fires. Instead, a producer task pulls chunks via the langsmith
           wrapper; our consumer await-s queue.get() under asyncio.timeout. On
           timeout, we explicitly close the producer's underlying SDK stream
           (sync stream.close()) and let the producer task complete naturally,
           which lets _TracedStream.__iter__ see normal StopIteration via the
           `else: self._end_trace()` branch (no error)."""
           import asyncio as _aio
           q: _aio.Queue = _aio.Queue(maxsize=64)
           SENTINEL = object()

           def _producer():
               # Sync producer — runs on the langsmith-wrapped stream's native
               # iterator. wrap_openai's _TracedStream.__iter__ runs to
               # completion (StopIteration → else: _end_trace()) when the
               # underlying stream closes normally.
               try:
                   for chunk in stream:
                       try:
                           q.put_nowait(chunk)
                       except _aio.QueueFull:
                           # Backpressure — block on the loop scheduler thread.
                           # Implementation detail: replace with an event-loop-aware put if needed.
                           pass
               finally:
                   try:
                       q.put_nowait(SENTINEL)
                   except Exception:
                       pass

           # Run the producer in the default executor so its for-loop cleanup
           # is decoupled from our async timeout.
           loop = _aio.get_running_loop()
           producer_fut = loop.run_in_executor(None, _producer)
           try:
               async with _aio.timeout(timeout_seconds):
                   while True:
                       chunk = await q.get()
                       if chunk is SENTINEL:
                           return
                       await on_chunk_async(chunk)
           except (_aio.TimeoutError, _aio.CancelledError):
               # Close the underlying SDK stream from OUTSIDE the producer's
               # for-loop. This makes the producer's `for chunk in stream` see
               # StopIteration cleanly (sync stream.close() is idempotent in
               # openai 2.28.0 / anthropic 0.97.0 via __exit__).
               try:
                   stream.close()
               except Exception:
                   logger.debug(
                       "stream.close() raised during Track A drain cancel — non-fatal",
                       exc_info=True,
                   )
               # Wait briefly for producer to complete; don't block forever.
               try:
                   await _aio.wait_for(_aio.wrap_future(producer_fut), timeout=2.0)
               except Exception:
                   logger.debug("producer await raised during Track A cancel — non-fatal", exc_info=True)
               raise
       ```
