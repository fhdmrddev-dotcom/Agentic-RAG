    ```typescript
    test.describe("063.1: thread switch mid-stream survives SSE", () => {
      test.beforeEach(async ({ page }) => { await signIn(page) })

      test("switch away + back: no blank window, single SSE per run, since>0 on reattach", async ({ page }) => {
        // requestLog with kind "active-runs" | "run-stream" | "post-message"
        type ReqKind = "active-runs" | "run-stream" | "post-message"
        const requestLog: Array<{ method: string; url: string; kind: ReqKind; phase: "started" | "finished" }> = []
        page.on("request", (req) => {
          const m = req.method(); const u = req.url()
          if (m === "POST" && /\/threads\/[^/]+\/messages$/.test(u)) requestLog.push({ method: m, url: u, kind: "post-message", phase: "started" })
          else if (m === "GET" && /\/threads\/[^/]+\/active-runs$/.test(u)) requestLog.push({ method: m, url: u, kind: "active-runs", phase: "started" })
          else if (m === "GET" && /\/runs\/[^/]+\/stream/.test(u)) requestLog.push({ method: m, url: u, kind: "run-stream", phase: "started" })
        })
        page.on("requestfinished", (req) => {
          if (req.method() === "GET" && /\/runs\/[^/]+\/stream/.test(req.url())) requestLog.push({ method: req.method(), url: req.url(), kind: "run-stream", phase: "finished" })
        })

        // Step 1: create thread A and send long prompt
        await createNewThread(page)
        const threadAUrl = page.url()
        await sendMessageInActiveThread(page, LONG_STREAM_PROMPT)
        await page.waitForTimeout(2_000)  // some tokens streamed; lastSeenOffsetRef populated

        // Step 2: create or switch to thread B
        await createNewThread(page)
        await page.waitForTimeout(2_000)

        // Step 3: switch back to thread A
        await page.goto(threadAUrl)

        // Assertion A: assistant bubble visible within 2s (no blank window)
        await expect(page.locator('[data-testid="assistant-message"]')).toBeVisible({ timeout: 2_000 })

        // Wait for any post-reattach activity to settle
        await page.waitForTimeout(2_000)

        // Assertion B: exactly ONE distinct run_id across all run-stream URLs
        const distinctRuns = new Set(
          requestLog
            .filter((r) => r.kind === "run-stream" && r.phase === "started")
            .map((r) => r.url.match(/\/runs\/([^/]+)\/stream/)?.[1])
            .filter(Boolean) as string[],
        )
        expect(distinctRuns.size).toBe(1)

        // Assertion C: since>0 on the reattach (the LAST run-stream request).
        // sendMessage's first call is "0"; reconcile-on-switch-back uses the
        // populated cursor.
        const sinceValues = requestLog
          .filter((r) => r.kind === "run-stream" && r.phase === "started")
          .map((r) => new URL(r.url).searchParams.get("since") ?? "")
        // First call: "0" (sendMessage). If a second call exists (reconcile reattach), it MUST be > "0".
        expect(sinceValues[0]).toBe("0")
        if (sinceValues.length > 1) {
          expect(sinceValues[sinceValues.length - 1]).not.toBe("0")
        }
      })
    })
    ```

    ```typescript
      test.beforeEach(async ({ page }) => { await signIn(page) })

      test("dispatching visibilitychange + focus in same tick: bubble visible, single SSE per run", async ({ page }) => {
        type ReqKind = "active-runs" | "run-stream"
        const requestLog: Array<{ url: string; kind: ReqKind; phase: "started" | "finished" }> = []
        page.on("request", (req) => {
          const u = req.url()
          if (req.method() === "GET" && /\/threads\/[^/]+\/active-runs$/.test(u)) requestLog.push({ url: u, kind: "active-runs", phase: "started" })
          else if (req.method() === "GET" && /\/runs\/[^/]+\/stream/.test(u)) requestLog.push({ url: u, kind: "run-stream", phase: "started" })
        })

        await createNewThread(page)
        await sendMessageInActiveThread(page, LONG_STREAM_PROMPT)
        await page.waitForTimeout(1_000)  // streaming begins; placeholder + SSE active

        // Force concurrent reconcile via synchronous event dispatch in the page context.
        // ChatArea.tsx (Phase 063 Plan 04) listens for BOTH events; in real
        // browsers tab activation can fire visibilitychange + focus in <50ms.
        await page.evaluate(() => {
          document.dispatchEvent(new Event("visibilitychange"))
          window.dispatchEvent(new Event("focus"))
        })

        // Wait for any reconcile activity to settle.
        await page.waitForTimeout(1_500)

        // Assertion A: assistant bubble visible (placeholder OR DB row — either is fine)
        await expect(page.locator('[data-testid="assistant-message"]')).toBeVisible({ timeout: 2_000 })

        // Assertion B: exactly ONE distinct run_id across all run-stream URLs.
        // The reconcileInFlightRef guard (Plan 04) ensures only ONE reconcile
        // body executes; the subscriptionsRef.has() short-circuit (Plan 02)
        // ensures only ONE subscribeToRun call fires.
        const distinctRuns = new Set(
          requestLog
            .filter((r) => r.kind === "run-stream")
            .map((r) => r.url.match(/\/runs\/([^/]+)\/stream/)?.[1])
            .filter(Boolean) as string[],
        )
        expect(distinctRuns.size).toBe(1)

        // Assertion C (anti-false-RED): at least ONE active-runs request fired
        // (proves the reconcile listener actually saw the dispatched events).
        const activeRunsCount = requestLog.filter((r) => r.kind === "active-runs").length
        expect(activeRunsCount).toBeGreaterThanOrEqual(1)
      })
    })
    ```

Asserting at least one active-runs request fired proves the reconcile listener fired at least once.
