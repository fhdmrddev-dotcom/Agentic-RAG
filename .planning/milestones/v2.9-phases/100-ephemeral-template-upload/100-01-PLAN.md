```python
import io, zipfile

def _make_ooxml(part_prefix: str, content_types_extra: str = "") -> bytes:
    bio = io.BytesIO()
    with zipfile.ZipFile(bio, "w", zipfile.ZIP_DEFLATED) as zf:
        zf.writestr("[Content_Types].xml",
            '<?xml version="1.0"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">'
            + content_types_extra + '</Types>')
        zf.writestr(f"{part_prefix}/document_marker.xml", "<x/>")
    return bio.getvalue()

@pytest.fixture
def valid_docx_bytes() -> bytes:
    return _make_ooxml("word")

@pytest.fixture
def valid_pptx_bytes() -> bytes:
    return _make_ooxml("ppt")

@pytest.fixture
def valid_xlsx_bytes() -> bytes:
    return _make_ooxml("xl")

@pytest.fixture
def renamed_binary_bytes() -> bytes:
    # A fake PE/EXE header renamed .docx — must FAIL zipfile.is_zipfile.
    return b"MZ\x90\x00\x03\x00\x00\x00" + b"\x00" * 64

@pytest.fixture
def oversized_ooxml_bytes() -> bytes:
    # Valid ZIP container padded past the 10 MB workspace_files_size_limit CHECK.
    base = _make_ooxml("word")
    return base + b"\x00" * (10 * 1024 * 1024 + 1 - len(base))
```
