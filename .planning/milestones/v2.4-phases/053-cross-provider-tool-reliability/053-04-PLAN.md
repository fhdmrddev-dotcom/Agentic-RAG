"""Unit tests for structured tool call parser — TOOL-03."""
from __future__ import annotations

import json

import pytest

from app.services.tool_parser import parse_structured_tool_calls, ToolCall, FunctionCall


# ---------------------------------------------------------------------------
# Happy path: markdown JSON blocks
# ---------------------------------------------------------------------------

def test_markdown_single_tool():
    text = '```json\n{"tool": "search_documents", "arguments": {"query": "hello world"}}\n```'
    calls = parse_structured_tool_calls(text, known_tools={"search_documents"})
    assert len(calls) == 1
    assert calls[0].function.name == "search_documents"
    assert json.loads(calls[0].function.arguments) == {"query": "hello world"}


def test_markdown_multiple_tools():
    text = (
        '```json\n{"tool": "search_documents", "arguments": {"query": "a"}}\n```\n'
        '```json\n{"tool": "read_document", "arguments": {"document_id": "123"}}\n```'
    )
    calls = parse_structured_tool_calls(text, known_tools={"search_documents", "read_document"})
    assert len(calls) == 2
    assert calls[0].function.name == "search_documents"
    assert calls[1].function.name == "read_document"


def test_markdown_with_surrounding_text():
    text = 'Let me search.\n\n```json\n{"tool": "search_documents", "arguments": {"query": "test"}}\n```\n\nDone.'
    calls = parse_structured_tool_calls(text, known_tools={"search_documents"})
    assert len(calls) == 1
    assert calls[0].function.name == "search_documents"


# ---------------------------------------------------------------------------
# Happy path: inline JSON
# ---------------------------------------------------------------------------

def test_inline_single_tool():
    text = '{"tool": "search_documents", "arguments": {"query": "inline"}}'
    calls = parse_structured_tool_calls(text, known_tools={"search_documents"})
    assert len(calls) == 1
    assert calls[0].function.name == "search_documents"

    """If markdown blocks exist, inline should be ignored."""
    text = (
        '```json\n{"tool": "search_documents", "arguments": {"query": "markdown"}}\n```\n'
        '{"tool": "read_document", "arguments": {"document_id": "inline"}}'
    )
    calls = parse_structured_tool_calls(text, known_tools={"search_documents", "read_document"})
    assert len(calls) == 1
    assert calls[0].function.name == "search_documents"

"""Unit tests for OpenRouter quality enhancements in streaming chat — TOOL-01."""
from __future__ import annotations

from unittest.mock import MagicMock, patch

import pytest

from app.services.openai_service import create_adaptive_streaming_chat, CallingMode


def _make_user_settings(
    provider: str = "openrouter",
    llm_model: str = "openrouter/gpt-4o",
    strategy: str = "quality",
):
    from types import SimpleNamespace
    return SimpleNamespace(
        active_provider=provider,
        llm_model=llm_model,
        openrouter_tool_strategy=strategy,
    )


# ---------------------------------------------------------------------------
# parallel_tool_calls=False
# ---------------------------------------------------------------------------

@patch("app.services.openai_service.get_llm_client")
@patch("app.services.openai_service._resolve_max_tokens")
@patch("app.services.openai_service._uses_max_completion_tokens")
def test_parallel_tool_calls_false(mock_uses_max, mock_resolve_tokens, mock_get_client):
    mock_client = MagicMock()
    mock_get_client.return_value = mock_client
    mock_resolve_tokens.return_value = 1000
    mock_uses_max.return_value = False
    
    user_settings = _make_user_settings()
    messages = [{"role": "user", "content": "hello"}]
    
    create_adaptive_streaming_chat(
        messages=messages,
        tool_choice="auto",
        user_settings=user_settings,
    )
    
    call_kwargs = mock_client.chat.completions.create.call_args.kwargs
    assert call_kwargs.get("parallel_tool_calls") is False
