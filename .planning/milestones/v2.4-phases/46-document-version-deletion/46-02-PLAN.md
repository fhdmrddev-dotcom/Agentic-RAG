```typescript
export async function deleteDocument(id: string, scope?: "version" | "all"): Promise<void> {
  const headers = await getAuthHeaders()
  const url = scope
    ? `${API_BASE}/documents/${id}?scope=${scope}`
    : `${API_BASE}/documents/${id}`
  const res = await fetch(url, { method: "DELETE", headers })
  if (!res.ok) throw new Error("Failed to delete document")
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user click → deleteDocument API call | Scope value originates from UI button choice, not user-typed text |
| frontend state → optimistic document removal | Filtering by filename for scope=all relies on local state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-46-F01 | Tampering | scope param in URL | accept | Scope value is hardcoded in button onClick handlers ("version" or "all") — not taken from user text input; no injection risk |
| T-46-F02 | Spoofing | deleteDocument API call | mitigate | `getAuthHeaders()` includes the Supabase JWT on every request; backend enforces ownership via user_id filter — frontend auth is defense in depth |
| T-46-F03 | Denial of Service | double-click delete | mitigate | `disabled={deleting}` on all footer buttons during in-flight request prevents duplicate submissions |
| T-46-F04 | Information Disclosure | error message text | accept | Error is "Delete failed. Please try again." — no backend detail leaked to UI |
| T-46-F05 | Tampering | optimistic filename-based removal (scope=all) | accept | Optimistic removal is local UI state only; Realtime DELETE events from Supabase provide ground truth; a wrong optimistic filter self-corrects on next loadDocuments() |
</threat_model>
