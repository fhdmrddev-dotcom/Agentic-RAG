Re-embed job shape (D-04/D-05 — all seams verified; resize_embedding_column + match filter live after Plan 04):
```python
async def reembed_job(supabase, user_id, app_settings, dims_changed):
    current = app_settings.embedding_model or "text-embedding-3-small"
    if dims_changed:  # only on a TRUE dims change
        await run_in_threadpool(lambda: supabase.rpc(
            "resize_embedding_column", {"new_dim": app_settings.embedding_dimensions}).execute())
    while True:  # resumable: same stale predicate each pass
        batch = await run_in_threadpool(lambda: supabase.table("document_chunks")
            .select("id, content").eq("user_id", user_id)          # RLS scope (V4)
            .neq("embedding_model", current).limit(BATCH).execute())  # IS DISTINCT FROM current
        if not batch.data:
            break
        vectors = await run_in_threadpool(embed_texts,
            [r["content"] for r in batch.data], user_settings=app_settings)
        for row, vec in zip(batch.data, vectors):  # write replacement THEN tag (D-05 non-destructive)
            await run_in_threadpool(lambda: supabase.table("document_chunks")
                .update({"embedding": vec, "embedding_model": current,
                         "embedding_dimensions": app_settings.embedding_dimensions})
                .eq("id", row["id"]).execute())
```
