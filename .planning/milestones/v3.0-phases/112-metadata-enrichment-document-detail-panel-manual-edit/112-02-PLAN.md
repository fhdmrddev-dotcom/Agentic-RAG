    # Phase 112 D-03 — re-extract precedence guard:
    # preserve any field a human marked _source='user' across re-extraction.
    # MUST read the PRIOR doc's _source map (NOT body.field) so ALL three re-extract
    # entry points (/upload, /reingest, /reextract) inherit it — they funnel through
    # this single write site (Pitfall 1).
    prior = supabase.table("documents").select("metadata") \
        .eq("id", document_id).maybe_single().execute().data  # sync — already in BackgroundTask thread
    prior_meta = (prior or {}).get("metadata") or {}
    user_fields = (prior_meta.get("_source") or {})  # {field: "user"}
    if user_fields:
        metadata_dict = metadata_dict or {}          # Pitfall 2: degrade returns None → promote to {} so it does NOT wipe edits
        preserved_source = metadata_dict.setdefault("_source", {})
        for fld, src in user_fields.items():
            if src != "user":
                continue
            if fld in prior_meta:
                metadata_dict[fld] = prior_meta[fld]   # restore the human value
            else:
                metadata_dict.pop(fld, None)           # human cleared it → keep it cleared
            preserved_source[fld] = "user"             # keep the marker
            # a human override has no model score → drop any fresh _confidence for it
            if isinstance(metadata_dict.get("_confidence"), dict):
                metadata_dict["_confidence"].pop(fld, None)
    # (the existing UPDATE at ~:1570 then writes the merged metadata_dict)
