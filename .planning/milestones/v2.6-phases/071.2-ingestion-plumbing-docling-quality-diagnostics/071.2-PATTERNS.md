```python
def legacy_text(raw: bytes, mime: str) -> tuple[str, str | None]:
    """pypdf + python-docx — fast, plain text only, no markdown."""
    if mime == PDF_MIME:
        from pypdf import PdfReader
        import io
        reader = PdfReader(io.BytesIO(raw))
        text = "\n\n".join(p.extract_text() or "" for p in reader.pages)
    elif mime == DOCX_MIME:
        from docx import Document as DocxDocument
        import io
        doc = DocxDocument(io.BytesIO(raw))
        text = "\n\n".join(p.text for p in doc.paragraphs if p.text.strip())
    else:
        raise ValueError(f"legacy_text: unsupported mime {mime!r}")
    return text, None  # legacy never produces markdown

def docling_text(raw: bytes, mime: str) -> tuple[str, str | None]:
    """Reuse the existing DoclingExtractor singleton — returns (text, full_markdown).

    Strict text mode: only returns text fields; tables/images/equations are
    aspects fetched by other registry entries. The temp-file dance is the
    same shape as DoclingExtractor.extract (167-238).
    """
    from app.services.extractors.docling import _get_converter
    import tempfile, os
    suffix = ".pdf" if mime == PDF_MIME else ".docx"
    with tempfile.NamedTemporaryFile(suffix=suffix, delete=False) as tf:
        tf.write(raw); tmp = tf.name
    try:
        result = _get_converter().convert(tmp)
        doc = result.document
        try:
            full_markdown = doc.export_to_markdown()
        except Exception:
            full_markdown = None
        text = full_markdown or doc.export_to_text()
        return text, full_markdown
    finally:
        try: os.unlink(tmp)
        except Exception: pass
```

```python
def extract_composable(
    raw: bytes, mime: str,
    engines: dict[str, str] | None = None,
) -> ExtractedDocument:
    """Phase 071.2 D-071.2-01 — per-aspect engine dispatch.

    `engines` keys (all optional, all default to app_settings.extraction.*):
        text:        'legacy' | 'docling' | 'pymupdf'
        tables:      'pdfplumber' | 'docling_tf'
        images:      'pdfplumber' | 'pymupdf_full' | 'docling_pictures'   (for PDF)
        docx_imgs:   'inline_shapes' | 'zip_xpath'                        (for DOCX)
        equations:   'none' | 'docling_formula'

    When `engines is None`, reads defaults from app_settings.extraction.*.
    Per-call override via /upload and /reextract `?engines=` query param.
    """
    from app.services.extractors.aspects import (
        TEXT_ENGINES, TABLE_ENGINES,
        IMAGE_ENGINES_PDF, IMAGE_ENGINES_DOCX, EQUATION_ENGINES,
    )
    from app.models.user_settings import load_app_settings

    settings = load_app_settings()
    eng = dict(engines or {})

    # Resolve defaults from app_settings.extraction.*
    eng.setdefault("text",
        settings.extraction_text_engine_pdf if mime == PDF_MIME
        else settings.extraction_text_engine_docx)
    eng.setdefault("tables",
        settings.extraction_table_engine_pdf if mime == PDF_MIME
        else "python-docx")  # DOCX tables always python-docx (D-071.2-02)
    eng.setdefault("images",
        settings.extraction_image_engine_pdf if mime == PDF_MIME
        else settings.extraction_image_engine_docx)
    eng.setdefault("equations", settings.extraction_equation_engine)

    # Dispatch each aspect through its registry
    text_fn = TEXT_ENGINES[eng["text"]]
    text, full_markdown = text_fn(raw, mime)

    tables = ()
    table_error = None
    try:
        if mime == PDF_MIME:
            tables = tuple(TABLE_ENGINES[eng["tables"]](raw, mime))
        elif mime == DOCX_MIME:
            # python-docx path lives in LegacyExtractor._extract_tables — reuse
            tables = tuple(_LEGACY._extract_tables(raw, mime))
    except Exception as exc:
        table_error = str(exc)
        log.warning("Composable table extraction failed: %s", exc)

    images = ()
    image_error = None
    try:
        if mime == PDF_MIME:
            images = tuple(IMAGE_ENGINES_PDF[eng["images"]](raw))
        elif mime == DOCX_MIME:
            images = tuple(IMAGE_ENGINES_DOCX[eng["images"]](raw))
    except Exception as exc:
        image_error = str(exc)
        log.warning("Composable image extraction failed: %s", exc)

    try:
        equations = tuple(EQUATION_ENGINES[eng["equations"]](raw, mime))
    except Exception as exc:
        log.warning("Composable equation extraction failed: %s", exc)
        equations = ()

    return ExtractedDocument(
        text=text,
        tables=tables,
        images=images,
        table_extraction_error=table_error,
        image_extraction_error=image_error,
        full_markdown=full_markdown,
        extractor_name=f"composable[{eng['text']}/{eng['tables']}/{eng['images']}/{eng['equations']}]",
        equations=equations,
    )
```

```python
def _parse_engines_hint(s: str | None) -> dict[str, str] | None:
    if not s:
        return None
    if not load_app_settings().extraction_per_call_hints_enabled:
        return None  # admin-disabled
    out: dict[str, str] = {}
    for pair in s.split(","):

        k, v = pair.split(":", 1)

        if k in ("text", "tables", "images", "equations") and v:

    return out or None
```
