export interface TechnicalNamesValue { showTechnical: boolean; toggle: () => void; setShowTechnical: (v: boolean) => void }

export function useTechnicalNames(): TechnicalNamesValue           // THROWS outside provider (citationNav idiom) — for writers (ControlRoom, Settings toggle)

// localStorage key: "technical-names"  ·  default OFF (plain)

export interface Term { plain: string; helper?: string; technical: string }

export type TermKey = keyof typeof TERM_MAP
